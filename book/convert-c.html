27 A File of C Written as Zeno

zeno convert reads C as well as Python. The promise is the one of the last chapter: what it writes is never silently wrong. What it can say in Zeno it says in Zeno, with the numbers C would give, bit for bit, and what it cannot it keeps as C, in an inline c block that the Zeno calls.

27.1 A first file

Here is a small meter, in C:

meter.c

/* A meter: readings, an average, and a checksum. */
#include <stdio.h>
#include <stdint.h>

#define COUNT 5

struct Reading {
    const char *name;
    int value;
};

uint8_t checksum(const uint8_t *bytes, int count) {
    uint8_t sum = 0;
    for (int i = 0; i < count; i++) {
        sum += bytes[i];
    }
    return (uint8_t)~sum;
}

int average(const struct Reading readings[], int count) {
    int total = 0;
    for (int i = 0; i < count; i++) {
        total += readings[i].value;
    }
    return total / count;
}

void limits(const struct Reading readings[], int count, int *low, int *high) {
    *low = readings[0].value;
    *high = readings[0].value;
    for (int i = 1; i < count; i++) {
        if (readings[i].value < *low) *low = readings[i].value;
        if (readings[i].value > *high) *high = readings[i].value;
    }
}

uint32_t tick(void) {
    static uint32_t count = 0;
    return ++count;
}

int main(void) {
    struct Reading readings[COUNT] = {
        {"volts", 230}, {"amps", 12}, {"hertz", 50}, {"watts", 2760}, {"temp", -4},
    };
    for (int i = 0; i < COUNT; i++) {
        printf("%-6s %5d\n", readings[i].name, readings[i].value);
    }
    int low, high;
    limits(readings, COUNT, &low, &high);
    printf("average %d, low %d, high %d\n", average(readings, COUNT), low, high);
    uint8_t frame[] = {0x10, 0x20, 0xF0};
    printf("checksum 0x%02X\n", checksum(frame, sizeof frame));
    uint32_t first = tick();
    printf("tick %u, tick %u\n", first, tick());
    return 0;
}

One command writes meter.zeno beside it:

$ zeno convert meter.c
Wrote meter.zeno from meter.c.
  9 pieces: 8 written as Zeno, 1 kept as C.

  Kept as C, in the `inline c` block:
    line 38, function `tick`: `static count`, a variable inside a function that keeps its value from one call to the next (line 39)

  Run it with `zeno run meter.zeno`. It prints what the C program prints,
  but for the notes above.

This is the Zeno it wrote:

$ cat meter.zeno
# kept: 1
# A meter: readings, an average, and a checksum.

import c

let c_part = inline c {
    #include <stdio.h>
    #include <stdint.h>
    #define COUNT 5
    struct Reading {
        const char *name;
        int value;
    };

    uint32_t tick(void) {
        static uint32_t count = 0;
        return ++count;
    }
}

# C's `/` of two whole numbers takes the quotient toward zero, where Zeno's `//`
# takes it down: -7 / 2 is -3 in C, and -7 // 2 is -4.
function c_divide(a: int, b: int) -> int {
    let down = a // b
    if down < 0 and down * b != a {
        return down + 1
    }
    return down
}

let COUNT = 5

class Reading {
    name: str
    value: int
}

function checksum(bytes: list[u8], count: int) -> u8 {
    var sum_: u8 = 0
    for i in range(0, count) {
        sum_ = u8.wrap(sum_ + bytes[i])
    }
    return u8.wrap(~sum_)
}

function average(readings: list[Reading], count: int) -> int {
    var total = 0
    for i in range(0, count) {
        total += readings[i].value
    }
    return c_divide(total, count)
}

function limits(readings: list[Reading], count: int, low: list[int], high: list[int]) {
    low[0] = readings[0].value
    high[0] = readings[0].value
    for i in range(1, count) {
        if readings[i].value < low[0] {
            low[0] = readings[i].value
        }
        if readings[i].value > high[0] {
            high[0] = readings[i].value
        }
    }
}

# `tick` is kept as C, in the block above.

function main() {
    let readings = [Reading("volts", 230), Reading("amps", 12), Reading("hertz", 50), Reading("watts", 2760), Reading("temp", -4)]
    for i in range(0, COUNT) {
        print(f"{readings[i].name:<6} {readings[i].value:5}")
    }
    var low = 0
    var high = 0
    let low_box = [low]
    let high_box = [high]
    limits(readings, COUNT, low_box, high_box)
    low = low_box[0]
    high = high_box[0]
    print(f"average {average(readings, COUNT)}, low {low}, high {high}")
    let frame: list[u8] = [0x10, 0x20, 0xF0]
    print(f"checksum 0x{checksum(frame, 3):02X}")
    let first: u32 = u32(c_part.tick())
    print(f"tick {first}, tick {c_part.tick()}")
}

main()

tick keeps a count from one call to the next in a static variable, which Zeno has no word for, so it stays C and main calls it as c_part.tick(). The rest is Zeno. The struct is a class and the arrays are lists. sum += bytes[i] on a uint8_t goes round at 256 as it does in C, written u8.wrap. total / count takes the quotient toward zero as C does, through a short helper with a comment that says why. The two pointers that limits writes through are lists of one: the caller puts low in low_box, makes the call, and takes it out again.

Both print the same:

$ zeno run meter.zeno
volts    230
amps      12
hertz     50
watts   2760
temp      -4
average 609, low -4, high 2760
checksum 0xDF
tick 1, tick 2
$ cc -o meter meter.c
$ ./meter
volts    230
amps      12
hertz     50
watts   2760
temp      -4
average 609, low -4, high 2760
checksum 0xDF
tick 1, tick 2

27.2 The numbers are C’s

The converter reads each type as C does on a 64-bit machine, and writes the type of Zeno that gives the same sums:

C Zeno
uint8_t to uint64_t u8 to u64, with wrap where C goes round
char, short, int, long int, and a store into a smaller type is checked
size_t int, where the value is never below zero; else kept
float, double f32, float
char * holding a text str
struct a class
an array a list

A signed sum that goes past the top is undefined in C, so no answer is C’s: Zeno stops with OverflowError, and the report says where a store into a smaller type is checked. / and % of whole numbers that may be below zero go through c_divide and c_remainder, which take the quotient toward zero; when both sides are never below zero they are // and %.

A hash of 64 bits reads as it does in C:

hash ^= u64(u8(letter_code(text, i)))
hash = u64.wrap(hash * FNV_PRIME)

A size_t is an int: it counts bytes and items, and stands beside ints on nearly every line. It is never past 263; if it were, Zeno would stop with OverflowError where C goes on.

27.3 Structs and pointers

A struct is a value in C and a class is shared in Zeno. So the converter writes a copy, a.copy(), wherever C copies: b = a;, a struct put into an array, or given to a function that changes it. Where nothing can tell, as for a struct a function only reads, it writes none.

Pointers are of a few kinds, and the converter writes four of them:

A pointer that Is written as
is indexed, or is an array parameter a list
points to one number written through a list of one
points to a struct the object
is const char * a text

Everything else a pointer does is kept: a pointer variable, pointer sums, NULL, malloc, a list made of next pointers. C does those well, and they have no Zeno that would read as Zeno.

27.4 What is kept, and what crosses

C can call only C, so a function that kept C calls is kept too. A variable lives on one side, so a global that kept C uses is kept, and so is Zeno that uses it. The Zeno calls the block with numbers, floats, bools and texts that C does not change; a call that would give it a struct or an array is kept. The block gets every #include, #define, struct and enum of the file, so the kept C is compiled as it was.

A line of the preprocessor that chooses (#if, #ifdef) keeps the whole file, since it is two programs and the converter would have to pick one. A main(int argc, char **argv) is kept and given the program’s arguments, so zeno run tool.zeno a b gives it what ./tool a b would.

27.5 What it does not promise

What C leaves undefined: a signed sum past the top and an index past the end make Zeno stop, and a variable C left without a value starts at 0. The order C works out the arguments of one call in is left to the compiler; Zeno’s is left to right. long is read as 64 bits, as everywhere but Windows. The rest is in docs/convert-c.md.

27.6 How it is checked

The programs of C in tools/convert are each built with cc and run, converted and run with zeno run, and what the two print and the status they end with are compared. Among them are a ring buffer, a frame parser and a state machine as they are written for a board, and all of that is written as Zeno, with nothing kept. As for Python, the converter reads what it wrote with Zeno’s own checker before it writes the file.
